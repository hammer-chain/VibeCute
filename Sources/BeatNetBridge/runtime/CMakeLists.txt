cmake_minimum_required(VERSION 3.14)
project(CaptionBeatNet C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${GGML_SOURCE} ggml)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../vendor)
add_library(caption_beatnet SHARED Runtime.cpp ${VENDOR}/src/common.cpp ${VENDOR}/src/compute.cpp ${VENDOR}/src/features.cpp ${VENDOR}/src/model.cpp ${VENDOR}/src/particle_filter.cpp)
target_include_directories(caption_beatnet PRIVATE ../include ${VENDOR}/src ${VENDOR}/third_party)
target_link_libraries(caption_beatnet PRIVATE ggml)
set_target_properties(caption_beatnet PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
add_executable(beatnet-parity Parity.cpp ../BeatNetBridge.c ${VENDOR}/src/stream.cpp)
target_include_directories(beatnet-parity PRIVATE ../include ${VENDOR}/src ${VENDOR}/third_party)
target_link_libraries(beatnet-parity PRIVATE caption_beatnet)
